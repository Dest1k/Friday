def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def reference(value):

def read_ref(value, *, project_source=False):
    """Stable owned no-follow read; 0664 history is authenticated, never chmodded."""
    reference(value)
    path=Path(value['path'])
    need(path.is_absolute() and path.resolve()==path,'reference_link')
    parents={}
    protected_anchor=False
    for parent in reversed(path.parents):
        s=parent.lstat()
        # Frozen project trees may retain 0775 directories below a private
        # owned ancestor. No outside group member can traverse that ancestor.
        permissive_project=(project_source and protected_anchor and s.st_uid==os.getuid()
                            and s.st_gid==os.getgid() and not s.st_mode&0o002)
        need(stat.S_ISDIR(s.st_mode) and s.st_uid in (0,os.getuid())
             and (not s.st_mode&0o022 or permissive_project),'protected_parent')
        protected_anchor=protected_anchor or (s.st_uid==os.getuid() and not s.st_mode&0o077)
        parents[parent]=tuple(getattr(s,k) for k in PARENT_FIELDS)
    limit=(64 if project_source else 16)*1024**2
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK|os.O_CLOEXEC)
    try:
        first=os.fstat(fd)
        need(stat.S_ISREG(first.st_mode) and first.st_uid==os.getuid() and first.st_gid==os.getgid()
             and first.st_nlink==1 and stat.S_IMODE(first.st_mode) in (0o600,0o644,0o664)
             and first.st_size<=limit,'owned_reference_file')
        with os.fdopen(os.dup(fd),'rb') as stream:
            raw=stream.read(limit+1)
        last,named=os.fstat(fd),path.lstat()
        need(len(raw)<=limit and len(raw)==first.st_size
             and all(getattr(first,k)==getattr(last,k)==getattr(named,k) for k in FIELDS)
             and sha(raw)==value['sha256'],'reference_bytes_or_identity_changed')
    finally:
        os.close(fd)
    need(all(tuple(getattr(p.lstat(),k) for k in PARENT_FIELDS)==s for p,s in parents.items()),'parent_identity_changed')
    return raw


def decode(raw):
    def pairs(items):
        result={}
        for k,v in items:
            need(k not in result,'duplicate_json_key')
            result[k]=v
        return result
    return json.loads(raw,object_pairs_hook=pairs,parse_constant=lambda _: (_ for _ in ()).throw(Refused('nonfinite_json')))


def document(value):
    return decode(read_ref(value))


def pinned_module(value, name):
    namespace={'__file__':value['path'],'__name__':name}
    exec(compile(read_ref(value),value['path'],'exec'),namespace)
    return namespace

def current_core_proof(binding_ref):
    """Future sole-integrator path: real selected native verifier, no projection."""
    value=binding(binding_ref)

                 primitives_pin=value['primitives'])

    need(proof['current_marker']['path']==str(Path(HOME)/'FRIDAY-INSTALL.json'), 'current_marker_outside_HOME')

def current_core_CAS(proof,binding_ref):

    need(sys.executable==value['native']['python'],'selected_current_native_python_required')
    need(proof==document(value['accepted_core'])['proof'],'current_core_invocation_memo_changed')

def _dsh154_approved_join(expected, name, bridge):
    """Exact separately reviewed DSH extension, retaining every approved A0 byte."""
    if name != PRODUCT_NAMES[0]:
        return expected
    delta=document(_DSH154_DELTA)
    need(delta['schema']=='frw.Root.dsh154-product-delta.v1'
         and delta['order']==[PRODUCT_NAMES[0]] and len(delta['files'])==1,
         'exact_reviewed_dsh154_scope')
    original=document(delta['reviewed_original_delta'])
    review=document(delta['independent_review'])
    need(review.get('accepted') is True and review.get('open_HIGH_MEDIUM')==0
         and review.get('product_delta')==delta['reviewed_original_delta']
         and review.get('base_sha256')==original['base']['sha256']
         and review.get('after_sha256')==original['files'][0]['after']['sha256'],
         'independent_dsh154_original_review')
    before=read_ref(delta['base'],project_source=True)
    after=read_ref(delta['files'][0]['after'],project_source=True)
    need(delta['files'][0]['path']==name and delta['files'][0]['before_sha256']==delta['base']['sha256'],
         'dsh154_exact_original_preimage')
    reference(bridge); read_ref(bridge)
    token=repr(_DSH154_PARENT_BRIDGE).encode()
    need(before.count(token)==1 and after.count(token)==1,'dsh154_one_current_bridge')
    rebound_before=before.replace(token,repr(bridge).encode(),1)
    rebound_after=after.replace(token,repr(bridge).encode(),1)
    need(expected==rebound_before,'dsh154_preserves_all_approved_a0_and_other_bytes')
    return rebound_after

def _source156_declared_closure(package):
    """Authenticate the complete declared routing graph, including sibling helper."""
    import ast
    declared=package.get('a0_private_closure')
    need(type(declared) is dict and set(declared)=={'source','copies'},'source156_closure_shape')
    parent=_SOURCE156_PARENT_CLOSURE
    need(set(declared['source'])==set(declared['copies'])==set(parent['copies']),'source156_closure_names')
    self_path=Path(__file__).resolve()
    self_ref={'path':str(self_path),'sha256':sha(self_path.read_bytes())}
    need(declared['source']['current_core.py']==declared['copies']['current_core.py']==self_ref,'source156_actual_core_route')
    copies=declared['copies']
    routes=(('compose_source_delta.py', {'CORE': 'current_core.py'}), ('retained_runtime_current.py', {'ANCESTOR_ADAPTER': 'current_core.py'}), ('source_current_a0_adapter.py', {'CURRENT_CORE': 'current_core.py', 'RUNTIME_VERIFIER': 'retained_runtime_current.py'}), ('publisher_current.UNBOUND.py', {'CURRENT_SOURCE_ADAPTER': 'source_current_a0_adapter.py', 'RUNTIME83_VERIFIER': 'retained_runtime_current.py'}), ('root_use_current.py', {'PUBLISHER': 'publisher_current.UNBOUND.py'}))
    for name,keys in routes:
        expected=read_ref(parent['copies'][name]).decode()
        for key,target in keys.items():
            nodes=[n for n in ast.parse(expected).body if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id==key for t in n.targets)]
            need(len(nodes)==1,'source156_one_assignment_'+key)
            n=nodes[0];lines=expected.splitlines(keepends=True)
            expected=''.join(lines[:n.lineno-1])+key+' = '+repr(copies[target])+'\n'+''.join(lines[n.end_lineno:])
        need(copies[name]['path']==str(self_path.parent/name) and declared['source'][name]==copies[name]
             and read_ref(copies[name])==expected.encode(),'source156_exact_route_'+name)
    for name in copies:
        read_ref(copies[name]);read_ref(declared['source'][name])
        if name=='ordinary_primitives.py':
            need(copies[name]['path']==str(self_path.parent/name) and copies[name]['sha256']==parent['copies'][name]['sha256']
                 and declared['source'][name]==parent['source'][name],'source156_sibling_ordinary_primitives')
        elif name.startswith('dependencies/'):
            need(copies[name]==parent['copies'][name] and declared['source'][name]==parent['source'][name],'source156_unchanged_dependencies')
    return True
