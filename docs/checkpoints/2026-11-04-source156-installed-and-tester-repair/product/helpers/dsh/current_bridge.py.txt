import hashlib
from pathlib import Path

OUT = Path(__file__).resolve().parent
ADAPTER_SHA256 = 'c93f27b2f7a2304be7d1b5a38ec65c7659a6c0ec3c81355188ab2d0df2992dc2'
CANONICAL_JOIN_PATH = '__REDACTED_PRIVATE_ABSOLUTE_PATH_59bf367ca84f__'


def verify(marker_ref, marker, *, read, document, consumer_path):
    request_ref = marker['ordinary_source_update_request']; request = document(request_ref)
    package_ref = request['package']; package = document(package_ref)
    if package_ref['path'] != str(OUT / 'PACKAGE.json') or package['adapter'] != {'path': str(OUT / 'core_successor.py'), 'sha256': ADAPTER_SHA256}:
        raise ValueError('core_exact_package_and_adapter')
    pin = package['adapter']; raw = read(pin)
    if hashlib.sha256(raw).hexdigest() != pin['sha256']:
        raise ValueError('core_adapter_pin')
    ns = {'__file__': pin['path'], '__name__': '_exact_core_h_source_provenance'}
    exec(compile(raw, pin['path'], 'exec'), ns)
    installed_join = ns['HOME'] / ns['JOIN']
    if marker_ref['path'] != str(ns['HOME'] / 'FRIDAY-INSTALL.json'):
        raise ValueError('core_current_consumer_paths')
    join_sha = marker['plugin_files'][ns['JOIN']]
    installed_raw = read({'path': str(installed_join), 'sha256': join_sha})
    if str(consumer_path) != str(installed_join):
        # One finalized source/canonical path, independently authenticated by
        # native_entry._native_graph's frozen source result and current input.
        # Never normalize an arbitrary caller path into the allowed name.
        canonical_join = Path(CANONICAL_JOIN_PATH)
        if str(consumer_path) != str(canonical_join) or canonical_join.resolve() != canonical_join:
            raise ValueError('core_current_consumer_paths')
        canonical_raw = read({'path': str(canonical_join), 'sha256': join_sha})
        if canonical_raw != installed_raw:
            raise ValueError('core_canonical_consumer_bytes')
    p = ns['_primitives'](package['primitives'])
    tx = ns['transaction_path'](request_ref)
    proof = ns['verify_source_journal'](package_ref, request_ref, p['ref'](tx / 'committed.json'), primitives_pin=package['primitives'])
    if proof['current_marker'] != marker_ref or document(marker_ref) != marker:
        raise ValueError('core_final_current_marker_cas')
    return proof['retained'], marker_ref, marker
