def parent_custody(path):
    """Record mutable access boundaries, without directory timestamps."""
    path = Path(path)
    require(path.is_absolute() and path.resolve() == path, 'parent_link')
    rows = {}
    private_ancestor = False
    for p in reversed((path, *path.parents)):
        st = p.lstat()
        # Historical owner-created source trees may be group-writable below
        # an owner-only directory. The inaccessible ancestor supplies custody;
        # keep every exact directory identity/mode for subsequent CAS.
        require(stat.S_ISDIR(st.st_mode) and st.st_uid in (0, os.getuid())
                and not st.st_mode & 0o002
                and (not st.st_mode & 0o020
                     or private_ancestor and st.st_uid == os.getuid()
                     and st.st_gid == os.getegid()), 'parent_owner_or_permissions')
        rows[str(p)] = identity(st, PARENT_FIELDS)
        if st.st_uid == os.getuid() and not st.st_mode & 0o077:
            private_ancestor = True
    return rows

def native_lock_checked(source, *, fd=None):
    import fcntl
    proof = _primitives({'path': str(OUT / 'canonical_primitives.py'), 'sha256': PRIMITIVES_SHA256})['document'](NATIVE_LOCK_PROOF)
    path = HOME / 'hermes-agent' / '.hermes-update.lock'
    require(isinstance(source, Path) and source == HOME / 'hermes-agent'
            and source.resolve() == source and proof['source'] == str(source)
            and proof['path'] == str(path) and proof['name'] == path.name
            and proof['writes_allowed'] is False and proof['deletion_allowed'] is False,
            'exact_native_lock_scope')
    if fd is None:
        fd = globals().get('NATIVE_LOCK_FD')
    require(type(fd) is int and fd >= 0, 'exact_native_lock_fd_required')
    expected = proof['identity']
    first = os.fstat(fd)
    require(set(expected) == set(FIELDS) and identity(first) == expected
            and identity(path.lstat()) == expected and path.resolve() == path
            and stat.S_ISREG(first.st_mode) and stat.S_IMODE(first.st_mode) == 0o644
            and first.st_uid == os.getuid() and first.st_gid == os.getegid()
            and first.st_nlink == 1 and first.st_size == 36
            and fcntl.fcntl(fd, fcntl.F_GETFL) & os.O_ACCMODE == os.O_RDONLY,
            'exact_native_lock_identity')
    try:
        fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)
    except BlockingIOError:
        require(False, 'native_lock_held_elsewhere')
    raw = os.pread(fd, first.st_size + 1, 0)
    require(len(raw) == first.st_size and hashlib.sha256(raw).hexdigest() == proof['sha256']
            and identity(first) == identity(os.fstat(fd)) == identity(path.lstat()),
            'exact_native_lock_bytes_or_custody')
    return {'proof': NATIVE_LOCK_PROOF, 'identity': expected, 'sha256': proof['sha256'],
            'exclusive_lease': True}
