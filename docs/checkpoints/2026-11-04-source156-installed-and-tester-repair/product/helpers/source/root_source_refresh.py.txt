def pin(value):
    row = json.loads(value)
    if set(row) != {'path', 'sha256'} or not Path(row['path']).is_absolute():
        raise ValueError('exact absolute artifact pin required')
    return row

def load_adapter(package):
    ref = package['adapter']
    raw = Path(ref['path']).read_bytes()
    if hashlib.sha256(raw).hexdigest() != ref['sha256']:
        raise ValueError('adapter pin changed')
    namespace = {'__file__': ref['path'], '__name__': '_bound_source119_adapter'}
    exec(compile(raw, ref['path'], 'exec'), namespace)
    return namespace

def stopped_guard(args, p, request, operation, clock):
    expected_guard = p['document'](request['package'])['refresh_boundary']['guard_module']
    if type(expected_guard) is not dict or args.guard_module != expected_guard:
        raise ValueError('exact accepted Root guard module required')
    if (args.phase_clock != operation['original_clock']
            or args.phase_operation != clock['operation']
            or clock.get('operation_id', args.phase_operation) != args.phase_operation):
        raise ValueError('same original source phase clock and operation required')
    raw = p['pinned'](args.guard_module)
    namespace = {'__file__': args.guard_module['path'], '__name__': '_exact_Root_stopped_guard'}
    exec(compile(raw, args.guard_module['path'], 'exec'), namespace)
    # FIRST native initialization: guard factory owns selected117 clean bootstrap.
    guard = namespace['make_stopped_guard'](args.stop_input, args.stopped_evidence,
        phase_clock_ref=args.phase_clock, phase_operation=args.phase_operation, receipt_phase='after_apply')
    if not callable(guard):
        raise ValueError('actual stopped guard callback required')
    return guard

class RecoveryDeadlineExceeded(TimeoutError):
    pass

class GuardedCancellation:
    """Event-like cancellation checked by actual PM refresh before publication.

    Direct accepted PM implementation retains this callback at its final precommit
    check. No worker transport or forced process termination is introduced. Any
    publication that races expiry still remains UNKNOWN for Root reconciliation.
    """
    def __init__(self, guard, deadline):
        self.event = threading.Event()
        self.guard, self.deadline, self.failure = guard, deadline, None
    def set(self):
        self.event.set()
    def is_set(self):
        if self.event.is_set():
            return True
        try:
            if time.monotonic() >= self.deadline:
                raise TimeoutError('forward source refresh deadline exhausted')
            if self.guard(deadline_monotonic=self.deadline, cleanup=False) is not True:
                raise ValueError('actual current stopped guard refused')
            if time.monotonic() >= self.deadline:
                raise TimeoutError('current guard crossed forward deadline')
        except RecoveryDeadlineExceeded:
            raise
        except Exception as exc:
            self.failure = type(exc).__name__
            self.event.set()
        return self.event.is_set()

SOURCE_BUILD_IMPLEMENTATION = {'path': '__REDACTED_PRIVATE_ABSOLUTE_PATH_c7ab21720f36__', 'sha256': '9fc26b645e39241a4627d3f778f58cd9556573cd09e51939d9c4dd2470c8aec9'}
WEB_FRESHNESS_MAXIMUM_SECONDS = 60

class FrontendFreshnessAborted(RuntimeError):
    pass

def require_current_web(p, *, deadline_monotonic):
    """Refuse native publication until canonical web output has a current receipt."""
    if not math.isfinite(deadline_monotonic):
        raise ValueError('finite original forward deadline required')
    finish_by = min(deadline_monotonic, time.monotonic() + WEB_FRESHNESS_MAXIMUM_SECONDS)
    remaining = finish_by - time.monotonic()
    if remaining <= 0:
        raise FrontendFreshnessAborted('canonical web freshness deadline exhausted')
    previous = {sig: signal.getsignal(sig) for sig in (signal.SIGTERM, signal.SIGINT, signal.SIGALRM)}
    if signal.getitimer(signal.ITIMER_REAL)[0] > 0:
        raise ValueError('fresh frontend check process must have no inherited alarm')
    def abort(signum, _frame):
        # RuntimeError deliberately escapes source_product_current's OSError /
        # SubprocessError catch, letting subprocess.run kill and reap its child.
        reason = 'deadline exhausted' if signum == signal.SIGALRM else 'interrupted'
        raise FrontendFreshnessAborted('canonical web freshness ' + reason)
    try:
        for sig in previous:
            signal.signal(sig, abort)
        signal.setitimer(signal.ITIMER_REAL, remaining)
        p['pinned'](SOURCE_BUILD_IMPLEMENTATION)
        from hermes_cli import source_build
        if Path(source_build.__file__).resolve() != Path(SOURCE_BUILD_IMPLEMENTATION['path']):
            raise ValueError('exact canonical source build helper required')
        source = HOME / 'hermes-agent'
        if not source_build.source_product_current(source, 'web', source / 'hermes_cli/web_dist'):
            raise ValueError('canonical web build required before native source identity')
        if time.monotonic() >= finish_by:
            raise FrontendFreshnessAborted('canonical web freshness deadline exhausted')
    finally:
        signal.setitimer(signal.ITIMER_REAL, 0)
        for sig, handler in previous.items():
            signal.signal(sig, handler)
