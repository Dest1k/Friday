# NON-EXECUTABLE REVIEW EXCERPT; UNBOUND, NOT AN INSTALLATION OR AUTHORITY.
# Definitions below intentionally omit imports, trust anchors and dependency bindings.
# Parse/compile verification never executes this text.

# The allowlist of exact closure pins, principal, package pins and schema sets are omitted.
# UNBOUND_* names deliberately have no values. No closure or grant is published.

def validate_closed_no_run(reader,c,runner,closure_ref):
    """Factual sealed history. Current cursor and future authority are separate."""
    import os,stat
    need(closure_ref in CLOSED_NO_RUN_CLOSURES,'closed_no_run_authenticated_closure')
    doc=reader.document;read=reader.read;closure=doc(ref(closure_ref))
    need(type(closure)is dict and set(closure)==D2_CLOSURE_FIELDS and closure['schema']=='frw.Root.engineering-no-send-admission-reconciled.v1','closed_no_run_exact_closure_schema')
    need(all(closure[k]is True for k in ('admission_consumed_and_preserved','no_replay','no_run','no_send_or_receive_witnesses','not_a_future_execution_grant','once_claim_absent','ordinary_authority_preserved')),'closed_no_run_closure_flags')
    need(closure['effect_state']=='CANCELLED_BEFORE_NATIVE_RUN_CREATION_NO_SEND' and closure['product_result']=='NOT_RUN' and closure['tester_result']=='STOPPED_EXCESSIVE_PREWIRE_NO_FULL_WINDOW_REMAINING','closed_no_run_terminal_not_run')
    root=Path(UNBOUND_PRIVATE_STATE_ROOT);need(str(runner.store.root)==str(root),'closed_no_run_exact_state_root')
    admission_ref=ref(closure['admission']);path=Path(admission_ref['path'])
    need(path.parent==root and re.fullmatch('continue-[a-f0-9]{64}\\.json',path.name),'closed_no_run_native_admission_path')
    key=path.name[len('continue-'):-len('.json')];claim=root/'continuations'/key
    need(closure['once_directory_preserved']==str(claim),'closed_no_run_exact_once_path')
    admitted=doc(admission_ref);raw=c.safe_read(path);need(sha(raw)==admission_ref['sha256'],'closed_no_run_admission_current_bytes')
    plan=doc(ref(closure['plan']));need(admitted.get('plan')==plan and admitted.get('plan_sha256')==sha(c.encoded(plan)),'closed_no_run_admission_plan_join')
    need(admitted.get('schema')=='friday.tester.closed-unknown-admission.v1' and admitted.get('status')=='ONE_NEW_EFFECT_ADMITTED_NO_RETRY' and admitted.get('original_outcome')=='UNKNOWN' and admitted.get('original_replayed')is False and admitted.get('response_binding')=='UNBOUND','closed_no_run_original_unknown_not_replayed')
    rid=closure['run_id'];need(type(rid)is str and re.fullmatch('root-eng-[a-z0-9-]{1,50}',rid) and plan.get('new_run_id')==rid and plan.get('schema')=='friday.tester.continue-closed-unknown.v2' and plan.get('operator')=='Root','closed_no_run_original_native_plan')
    value=plan.get('current_tuple');need(type(value)is dict and set(value)=={'installed','layers','native','project','service_owner','source'} and value['installed']==7 and value['layers']==value['native']==0 and type(value['source'])is int and value['source']>151 and value['project']==value['source'] and value['service_owner']=='Root','closed_no_run_original_source_tuple')
    need(plan.get('new_task',{}).get('kind')=='ordinary-document-task' and plan['new_task'].get('source_reference')is None and len(plan.get('new_scenario',{}).get('steps',[]))==1,'closed_no_run_original_ordinary_plan')
    clock=closure['original_case_clock'];need(clock==plan.get('clock') and set(clock)=={'boot_id','deadline_monotonic','origin_monotonic'} and type(clock['boot_id'])is str and clock['boot_id'] and all(type(clock[k])in(int,float)for k in('deadline_monotonic','origin_monotonic')) and clock['deadline_monotonic']>clock['origin_monotonic'],'closed_no_run_original_clock_unchanged')
    cursor=ref(closure['current_cursor']);need(cursor['path']==str(root/'cursor.json') and plan['predecessor'].get('cursor_sha256')==cursor['sha256'] and sha(c.encoded(admitted.get('predecessor_cursor')))==cursor['sha256'],'closed_no_run_historical_cursor_custody')
    # cursor.json is mutable global progress. The sealed predecessor cursor above
    # remains authenticated after legitimate later receives advance that file.
    a=path.lstat();need(stat.S_ISREG(a.st_mode) and a.st_uid==os.getuid() and a.st_nlink==1 and stat.S_IMODE(a.st_mode)==0o600,'closed_no_run_admission_file_custody')
    s=claim.lstat();need(stat.S_ISDIR(s.st_mode) and s.st_uid==os.getuid() and stat.S_IMODE(s.st_mode)==0o700 and list(claim.iterdir())==[],'closed_no_run_empty_once_claim')
    run=runner.store.run_path(rid);need(not os.path.lexists(run) and not os.path.lexists(run.with_name(run.name+'.new')) and not os.path.lexists(claim/'SEND-ATTEMPT.json'),'closed_no_run_no_run_or_send_attempt')
    actual=doc(ref(closure['actual']));need(actual.get('process_reaped')is True and type(actual.get('exit'))is int and 0<=actual['exit']<=255,'closed_no_run_process_reaped')
    for name in ('stdout','stderr'):
        stream=actual.get(name);need(stream=='' or type(stream)is dict and read(ref(stream))==b'','closed_no_run_empty_process_streams')
    safe=doc(ref(closure['safe_stop_intent']));preserved=[safe[name]for name in('admission_preserved','native_admission_preserved')if name in safe]
    need(preserved and all(value==admission_ref for value in preserved) and safe.get('actual_case_clock')==clock and safe.get('native_run_absent_before_signal')is True and safe.get('native_send_or_reply_witness_absent_before_signal')is True and safe.get('no_replay')is True,'closed_no_run_zero_outbound_before_stop')
    resources=doc(ref(closure['resources_closed']));need(resources.get('Root_helpers_remaining')==[] and resources.get('owned_foreground_released')is True and resources.get('resources_zero')is True and resources.get('resource_outcome_known')is True and resources.get('original_clock_not_reset')is True and resources.get('product_acceptance')is False,'closed_no_run_resources_closed')
    boundary=doc(ref(resources['final_boundary']));need(boundary.get('actual')==closure['actual'] and boundary.get('native_admission')==admission_ref and boundary.get('native_run_absent')is True and boundary.get('native_send_or_reply_witnesses')==[] and boundary.get('once_claim_children')==[] and boundary.get('once_claim_directory')==str(claim) and boundary.get('own_timeout_and_native_process_gone')is True and boundary.get('resource_outcome_known')is True and boundary.get('product_result')=='NOT_RUN_NO_SEND' and boundary.get('cursor')==cursor,'closed_no_run_final_no_send_boundary')
    need(boundary.get('gateway',{}).get('result',{}).get('active_agents')==0 and boundary['gateway']['result'].get('active_work')is None and boundary.get('owned_projection',{}).get('result',{}).get('snapshot',{}).get('active')is False and type(boundary.get('scheduler'))is list and boundary['scheduler'] and all(row.get('num_reqs')==row.get('num_waiting_reqs')==row.get('num_pending_tokens')==row.get('num_tokens')==0 for row in boundary['scheduler']),'closed_no_run_resource_zero_witnesses')
    effect=dict(kind='no_run_admission',id=rid,reference=admission_ref)
    owner=boundary['owned_projection']['result']['snapshot'];need(type(owner.get('session_id'))is str and owner['session_id'],'closed_no_run_owner_session')
    sig=signature(plan['new_scenario']['steps'][0]);affected=sorted(['friday-session:'+owner['session_id'],UNBOUND_MODEL_RESOURCE,'telegram:'+PRINCIPAL['account_id']+':'+PRINCIPAL['chat_id'],'tester:'+str(root)]+([]if sig['input']is None else['fixture-sha256:'+sig['input']['sha256']]))
    return dict(effect=effect,row=admitted,plan=plan,closure=closure,closure_reference=closure_ref,admission=admission_ref,key=key,signature=sig,affected_resources=affected,same_tuple=True)

def closed_no_run_inventory(reader,c,runner):
    """Authenticate every sealed no-run receipt, regardless of next signature."""
    entries={};keys=set()
    for closure_ref in CLOSED_NO_RUN_CLOSURES:
        entry=validate_closed_no_run(reader,c,runner,closure_ref);rid=entry['effect']['id']
        need(rid not in entries and entry['key']not in keys,'closed_no_run_unique_inventory')
        entries[rid]=entry;keys.add(entry['key'])
    return entries

def validate(grant_ref,reader,c,runner,*,case_ref=None,scenario_ref=None,bindings_ref=None,run_id=None,plan=None,tester_package=None):
    """Rechecks actual native inventory, current custody and every collision.

    `reader` is the existing protected custody reader, never a model callback.
    A Root reconciliation says only that the old physical effect cannot continue;
    it neither changes old semantic UNKNOWN nor authorizes the new effect itself.
    """
    doc=reader.document;read=reader.read;grant=doc(ref(grant_ref))
    need(type(grant)is dict and set(grant)==GRANT_FIELDS and grant['schema']==SCHEMA and grant['operator']=='Root' and grant['accepted']is True,'exact_predeclared_Root_grant')
    need(grant['old_effects_and_clocks_preserved']is True and grant['new_attempt_only']is True and grant['no_retry']is True and grant['semantic_acceptance']=='NOT_ESTABLISHED','old_results_and_no_retry')
    need(type(grant['attempt_id'])is str and re.fullmatch('engineering-[a-z0-9-]{1,72}',grant['attempt_id']) and type(grant['new_run_id'])is str and re.fullmatch('root-eng-[a-z0-9-]{1,50}',grant['new_run_id']),'distinct_attempt_identity')
    for supplied,key in ((case_ref,'case'),(scenario_ref,'scenario'),(bindings_ref,'current_bindings'),(run_id,'new_run_id')):
        need(supplied is None or supplied==grant[key],'caller_'+key+'_join')
    case=doc(ref(grant['case']));scenario=doc(ref(grant['scenario']));need(len(scenario.get('steps',[]))==1,'single_step_only');sig=signature(scenario['steps'][0])
    need(any(row.get('scenario')==grant['scenario'] for row in case.get('scenario_steps',[])),'case_scenario_join')
    need(grant['principal']==PRINCIPAL and type(grant['session_id'])is str and grant['session_id'] and type(grant['session_key'])is str and grant['session_key'],'exact_owner_principal')
    bindings=doc(ref(grant['current_bindings']));source=doc(ref(bindings['installed_source']));package=doc(ref(bindings['package']));actual=doc(ref(bindings['actual_tuple']));services=doc(ref(bindings['permanent_services']));proof=doc(ref(bindings['fresh_native_proof']));writer=doc(ref(bindings['sole_writer']))
    need(bindings.get('schema')==BINDING_SCHEMA and (package.get('schema')==CANDIDATE_SCHEMA or (package.get('schema')==CURRENT_CANDIDATE_SCHEMA and bindings['package']==CURRENT_CANDIDATE_PACKAGE)) and package['target_source_receipt']==grant['candidate_source_receipt'],'installed_new_candidate')
    read(ref(grant['candidate_source_receipt']))
    native=source['native_continuity'];generation=grant['candidate_native_generation']
    need(type(generation)is str and re.fullmatch('[a-f0-9]{32}',generation) and generation!=UNBOUND_PRIOR_NATIVE_GENERATION and native['actual_native_generation']==generation==proof['actual_native_generation']==actual['actual_native_generation']==actual['native_generation'],'exact_new_native_generation')
    need(source['package']==proof['package']==bindings['package'] and source['fresh_native_proof_artifact']==bindings['fresh_native_proof'] and source['proof']==proof and services['installed_source']==actual['installed_source']==bindings['installed_source'] and actual['permanent_services']==bindings['permanent_services'],'installed_source_proof_services_join')
    need(actual.get('schema')=='frw.Root.actual-installed-tuple.v1' and type(actual['tuple']['source'])is int and actual['tuple']['source']>151 and actual['tuple']['project']==actual['tuple']['source'] and native==proof['native']==services['native'],'new_composition_tuple')
    need(services.get('accepted_service_readiness')is True and services.get('permanent_no_test_expiry')is True and source.get('runtime_or_user_semantics_accepted')is False and proof.get('runtime_or_user_semantics_accepted')is False,'readiness_not_semantic_acceptance')
    need(writer==dict(schema='frw.Root.s07-sole-writer-authority.v1',operator='Root',session_id=grant['session_id'],session_key=grant['session_key'],same_session_human_or_other_writer=False,no_new_worker_rights=True),'sole_writer_current_session')
    cursor=ref(grant['cursor']);need(cursor['path']==str(runner.store.root/'cursor.json') and sha(c.safe_read(cursor['path']))==cursor['sha256'],'current_cursor_CAS')
    resources=grant['affected_resources'];need(type(resources)is list and resources and resources==sorted(set(resources)) and all(type(x)is str and x for x in resources),'exact_nonempty_affected_resources')
    need(type(grant['collisions'])is list and grant['collisions'],'reconciled_collisions_required')
    known={}
    for closure_ref in grant['collisions']:
        closure=doc(ref(closure_ref))
        if closure.get('schema')=='frw.Root.engineering-no-send-admission-reconciled.v1':
            entry=validate_closed_no_run(reader,c,runner,closure_ref);rid=entry['effect']['id']
            need(rid not in known and rid!=grant['new_run_id'] and entry['signature']==sig and resources==entry['affected_resources'] and entry['plan']['current_tuple']==actual['tuple'],'closed_no_run_exact_same_tuple_missing_operation_only')
            known[rid]=entry
            continue
        need(type(closure)is dict and set(closure)==CLOSURE_FIELDS and closure['schema']==RECONCILED and closure['operator']=='Root' and closure['accepted']is True,'exact_Root_effect_reconciliation')
        effect=closure['effect'];need(type(effect)is dict and set(effect)=={'kind','id','reference'} and effect['kind']in('run','no_run_admission') and type(effect['id'])is str and effect['id'] not in known and effect['id']!=grant['new_run_id'],'unique_original_effect')
        old=doc(ref(effect['reference']));raw=c.safe_read(effect['reference']['path']);need(sha(raw)==effect['reference']['sha256'],'original_effect_current_bytes')
        if effect['kind']=='run':
            need(effect['reference']['path']==str(runner.store.run_path(effect['id'])) and old['id']==effect['id'],'original_native_run_identity')
            native_ref=old.get('continuation_reference');need(type(native_ref)is dict,'original_native_admission_required')
            admission=runner.read_reference(native_ref);old_plan=admission.get('plan')
            need(type(old_plan)is dict and admission.get('plan_sha256')==sha(c.encoded(old_plan)) and old_plan.get('new_run_id')==old['id'] and old_plan.get('new_scenario')==old['scenario'],'original_native_admission_join')
            old_steps=old['scenario']['steps'];need(closure['original_clock']==old_plan['clock'],'old_clock_unchanged')
        else:
            old_plan=old['plan'];need(old.get('schema')=='friday.tester.closed-unknown-admission.v1' and old.get('status')=='ONE_NEW_EFFECT_ADMITTED_NO_RETRY' and old['plan_sha256']==sha(c.encoded(old_plan)) and old_plan['new_run_id']==effect['id'] and not runner.store.run_path(effect['id']).exists(),'original_no_run_admission_identity')
            need(Path(effect['reference']['path']).parent==runner.store.root and Path(effect['reference']['path']).name.startswith('continue-'),'native_admission_path');old_steps=old_plan['new_scenario']['steps'];need(closure['original_clock']==old_plan['clock'],'orphan_clock_unchanged')
        need(any(signature(s)==sig for s in old_steps if ordinary(s)),'exact_original_signature')
        old_bindings=doc(ref(closure['old_current_bindings']));old_source=doc(ref(old_bindings['installed_source']));capture=doc(ref(closure['original_capture']));old_package=doc(ref(old_bindings['package']))
        need(old_source['package']==old_bindings['package'] and old_package['target_source_receipt']!=grant['candidate_source_receipt'] and old_source['native_continuity']['actual_native_generation']!=generation,'actual_version_changed')
        need(capture.get('run_id')==effect['id'] and capture.get('current_bindings')==closure['old_current_bindings'] and capture.get('source_identity')==old_source['native_continuity']['source_identity'],'original_capture_version_join')
        need(closure['original_semantic_outcome']in('PASS','FAIL','UNKNOWN','NOT_RUN') and closure['effect_state']in('COMPLETED','CANCELLED','ISOLATED') and closure['affected_resources']==resources and closure['no_remaining_effect_on_affected_resources']is True and closure['old_bytes_and_clock_preserved']is True and closure['new_effect_authorized']is False,'physical_effect_reconciled_without_semantic_rewrite')
        need(type(closure['evidence'])is list and closure['evidence'],'physical_evidence_required')
        for proof_ref in closure['evidence']:read(ref(proof_ref))
        known[effect['id']]=dict(effect=effect,row=old,plan=old_plan,closure=closure)
    own_run=None
    if runner.store.run_path(grant['new_run_id']).exists():
        own_run=runner.store.get(runner.store.run_path(grant['new_run_id']))
        need(plan is not None and own_run.get('status')=='PREPARED' and own_run.get('send_state')=='none' and own_run.get('send_attempts')==0 and own_run.get('scenario')==scenario,'new_attempt_already_consumed')
    actual_collisions=set()
    for path in runner.store.runs.glob('*.json'):
        old=runner.store.get(path)
        if own_run is not None and old==own_run:continue
        if any(signature(s)==sig for s in old.get('scenario',{}).get('steps',[])if ordinary(s)):
            need(old.get('id')in known and known[old['id']]['effect']['kind']=='run' and known[old['id']]['row']==old,'unreconciled_existing_run_collision');actual_collisions.add(old['id'])
    for path in runner.store.root.glob('continue-*.json'):
        admission=runner.store.get(path);old_plan=admission.get('plan',{});rid=old_plan.get('new_run_id');need(type(rid)is str,'valid_native_admission')
        if runner.store.run_path(rid).exists() or (plan is not None and old_plan==plan):continue
        if any(signature(s)==sig for s in old_plan.get('new_scenario',{}).get('steps',[])if ordinary(s)):
            need(rid in known and known[rid]['effect']['kind']=='no_run_admission' and known[rid]['row']==admission,'unreconciled_no_run_collision');actual_collisions.add(rid)
    need(actual_collisions==set(known),'complete_exact_collision_inventory')
    for path in runner.store.root.glob('continue-*.json'):
        admission=runner.store.get(path);admitted=admission.get('plan',{})
        if admitted.get('new_run_id')==grant['new_run_id']:
            need(plan is not None and admitted==plan and admission.get('plan_sha256')==sha(c.encoded(plan)) and admission.get('status')=='ONE_NEW_EFFECT_ADMITTED_NO_RETRY','new_attempt_admission_consumed')
    # Current bindings name the installed product. Timing policy acceptance names
    # the independently sealed tester package, including its exact caller.
    accepted=doc(ref(grant['future_timing_acceptance']))
    accepted_tester=doc(ref(accepted['production_package']))
    need(accepted['production_package']!=bindings['package'],'separate_product_and_tester_packages')
    need(tester_package is None or accepted_tester==tester_package,'caller_tester_package_join')
    policy,_=validate_policy(grant['future_timing_policy'],grant['future_timing_acceptance'],reader,package=accepted_tester,case_ref=grant['case'],scenario_ref=grant['scenario'],plan=plan)
    limits=grant['limits'];need(type(limits)is dict and set(limits)=={'scenario_seconds','ACK_seconds','observation_seconds','outer_seconds','parent_watchdog_seconds'} and all(type(v)is int and v>0 for v in limits.values()),'predeclared_finite_limits')
    need(limits==dict(scenario_seconds=policy['scenario_budget_seconds'],ACK_seconds=policy['ack_timeout_seconds'],observation_seconds=policy['reply_window_seconds'],outer_seconds=policy['outer_budget_seconds'],parent_watchdog_seconds=policy['parent_watchdog_seconds']),'exact_policy_limits')
    need(scenario['deadline_seconds']==limits['scenario_seconds'] and scenario['steps'][0]['observe']['window_seconds']==limits['observation_seconds'] and limits['ACK_seconds']+limits['observation_seconds']<limits['scenario_seconds']<limits['outer_seconds']<limits['parent_watchdog_seconds'],'limits_match_scenario')
    if plan is not None:
        need(plan['new_run_id']==grant['new_run_id'] and plan['new_scenario']==scenario and plan['current_tuple']==actual['tuple'] and plan.get('predecessor')==grant['predecessor'] and plan['ack_timeout_seconds']==limits['ACK_seconds'],'exact_native_plan_binding')
        need(plan['clock']['deadline_monotonic']==plan['clock']['origin_monotonic']+limits['outer_seconds'],'fresh_declared_clock')
    return dict(grant=grant,known=known,signature=sig,tuple=actual['tuple'],closed_no_run_inventory=closed_no_run_inventory(reader,c,runner))
